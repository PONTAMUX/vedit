cmake_minimum_required(VERSION 3.20)
project(vimg LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Test)
add_library(vimgcore STATIC src/model.cpp src/render.cpp src/mainwindow.cpp src/canvas.cpp src/panels.cpp src/dialogs.cpp)
target_include_directories(vimgcore PUBLIC src)
target_link_libraries(vimgcore PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets)
add_executable(vimg src/main.cpp)
target_link_libraries(vimg PRIVATE vimgcore)
set_target_properties(vimg PROPERTIES WIN32_EXECUTABLE ON)
add_executable(vimg_test tests/guitest.cpp)
target_link_libraries(vimg_test PRIVATE vimgcore Qt6::Test)
